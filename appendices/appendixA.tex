\chapter{Complete Antenna Expressions}\label{ap:antExpressions}

This appendix includes the complete expressions of the antennae computed using
\textit{AntCalc}. It shows the unintegrated and integrated antennae, along with
the commands used to compute them.

A note on conventions: the unintegrated antenna functions below are given exactly as
\textit{AntCalc} returns them, in terms of $s_{ij}$ and $q^2$. The integrated antennae,
however, are quoted at $q^2=1$, matching \texttt{IntegrateAntenna}'s default convention. This
does not break generality as established in Subsection~\ref{subsec:MIs} and Eq.~\eqref{eq:MIscaling}, a
quantity of this kind scales with $q^2$ as a single power fixed by its mass dimension and
$\epsilon$-scaling, so the complete $q^2$-dependent result is recovered by multiplying the
expression given here by $(q^2)^{\lambda+\kappa\epsilon}$ for the appropriate $\lambda$ and
$\kappa$. The explicit case of $\mathcal A_3^0$, which scales as $(q^2)^{-\epsilon}$, is worked
out in full in Eq.~\eqref{eq:A30withR30Sub}. We must also here introduce the shorthand,
\begin{equation}
    C_\epsilon \equiv \frac{(4\pi)^\epsilon e^{-\gamma_E\epsilon}}{8\pi^2}
    = \frac{1}{C(\epsilon,1)}
\end{equation}
as the one-loop normalisation factor.

\section{LO Antenna}

\subsection{$A_2^0$}
Running \texttt{BuildAntenna[A,2,0]}, we get the built result,
\begin{equation}
    A_2^0 = 1.
\end{equation}
By running \texttt{BuildAndIntegrateAntenna[A,2,0]} we get the integrated result, 
\begin{equation}
    \mathcal A_2^0 = \frac14 + \epsilon\frac98 - \epsilon^2\!\left(\frac{9}{16}-\frac{\pi^2}{48}\right).
\end{equation}

\section{NLO Antennae}

\subsection{$A_2^1$}
Running \texttt{BuildAntenna[A,2,1]}, we get the built result,
\begin{equation}
    A_2^1 = -\frac12(3+2\epsilon)B_0\!\left(q^2;0,0\right) - q^2C_0\!\left(0,0,q^2;0,0,0\right).
\end{equation}
By running \texttt{BuildAndIntegrateAntenna[A,2,1]} we get the integrated result, 
\begin{equation}
    \mathcal A_2^1 = -\frac1{\epsilon^2}-\frac3{2\epsilon}-4+\frac{7\pi^2}{12}+\epsilon\!\left(\frac{7\zeta_3}{3}-8+\frac{7\pi^2}8\right)+\epsilon^2\!\left(\frac{7\zeta_3}2-16+\frac{7\pi^2}3-\frac{73\pi^4}{1440}\right) + \mathcal O\!\left(\epsilon^3\right).
\end{equation}

\subsection{$A_3^0$}
Running \texttt{BuildAntenna[A,3,0]}, we get the built result,
\begin{equation}
    A_3^0 = \frac{2s_{12}^2+2s_{12}(s_{13}+s_{23})}{q^2s_{13}s_{23}}-\epsilon\frac{2}{q^2}+(1-\epsilon)\frac{s_{13}^2+s_{23}^2}{q^2s_{13}s_{23}}.
\end{equation}
By running \texttt{BuildAndIntegrateAntenna[A,3,0]} we get the integrated result, 
\begin{equation}
    \begin{gathered}
        \mathcal A_3^0 = \frac1{\epsilon^2} + \frac3{2\epsilon} + \frac{19}4 - \frac{7\pi^2}{12} + \epsilon\!\left(-\frac{25\zeta_3}{3}+\frac{109}8-\frac{7\pi^2}8\right)\\ 
        + \epsilon^2\!\left(-\frac{25\zeta_3}2+\frac{639}{16}-\frac{133\pi^2}{48}-\frac{71\pi^4}{1440}\right) + \mathcal O\!\left(\epsilon^3\right).
    \end{gathered}
\end{equation}

\section{NNLO Antennae}

\subsection{$A_2^2$}

Running \texttt{BuildAntenna[A,2,2,Component->Leading]}, we get the built result,
\begin{equation}
    A_2^2 = \frac{\mathcal A_2^2}{C(\epsilon,2)} = 4^{2\epsilon-3}\pi^{2\epsilon-4}e^{-2\gamma_E\epsilon}\,\mathcal A_2^2,
\end{equation}
where $\mathcal A_2^2$ is the integrated result given below, shown through $\mathcal O(\epsilon^0)$, and $C(\epsilon,2)=G_2/S_\epsilon^2$.

By running \texttt{BuildAndIntegrateAntenna[A,2,2,Component->Leading]} we get the integrated result, 
\begin{equation}
    \begin{gathered}
        \mathcal A_2^2 = \frac1{4\epsilon^4}+\frac{17}{8\epsilon^3}+\frac1{\epsilon^2}\!\left(\frac{433}{144}-\frac{\pi^2}2\right)+\frac1\epsilon\!\left(\frac{4045}{864}-\frac{83\pi^2}{48}+\frac{7\zeta_3}{12}\right)\\
        - \frac{9083}{5184}-\frac{2153\pi^2}{864}+\frac{263\pi^4}{1440}+\frac{13\zeta_3}{9}+\mathcal O(\epsilon).
    \end{gathered}
\end{equation}
