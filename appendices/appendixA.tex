\chapter{Complete Antenna Expressions}\label{ap:antExpressions}

This appendix includes the complete expressions of the antennae computed using
\textit{AntCalc}. It shows the unintegrated and integrated antennae, along with 
the commands used to compute them.

\section{LO Antenna}

\subsection{$A_2^0$}
Running \texttt{BuildAntenna[A,2,0]}, we get the built result,
\begin{equation}
    A_2^0 = 1.
\end{equation}
By running \texttt{BuildAndIntegrateAntenna[A,2,0]} we get the integrated result, 
\begin{equation}
    \mathcal A_2^0 = \frac14 + \epsilon\frac98 - \epsilon^2\!\left(\frac{9}{16}-\frac{\pi^2}{48}\right).
\end{equation}

\section{NLO Antennae}

\subsection{$A_2^1$}
Running \texttt{BuildAntenna[A,2,0]}, we get the built result,
\begin{equation}
    A_2^1 = -\frac12(3+2\epsilon)B_0\!\left(q^2;0,0\right) - q^2C_0\!\left(0,0,q^2;0,0,0\right).
\end{equation}
By running \texttt{BuildAndIntegrateAntenna[A,2,1]} we get the integrated result, 
\begin{equation}
    \mathcal A_2^1 = -\frac1{\epsilon^2}-\frac3{2\epsilon}-4+\frac{7\pi^2}{12}+\epsilon\!\left(\frac{7\zeta_3}{3}-8+\frac{7\pi^2}8\right)+\epsilon^2\!\left(\frac{7\zeta_3}2-16+\frac{7\pi^2}3-\frac{73\pi^4}{1440}\right) + \mathcal O\!\left(\epsilon^3\right).
\end{equation}

\subsection{$A_3^0$}
Running \texttt{BuildAntenna[A,3,0]}, we get the built result,
\begin{equation}
    A_3^0 = \frac{2s_{12}^2-2s_{12}(s_{13}+s_{23})}{q^2s_{13}s_{23}}+\epsilon\frac{2}{q^2}+(\epsilon-1)\frac{s_{13}^2+s_{23}^2}{q^2s_{13}s_{23}}.
\end{equation}
By running \texttt{BuildAndIntegrateAntenna[A,3,0]} we get the integrated result, 
\begin{equation}
    \begin{gathered}
        \mathcal A_3^0 = \frac1{\epsilon^2} + \frac3{2\epsilon} + \frac{19}4 - \frac{7\pi^2}{12} + \epsilon\!\left(-\frac{25\zeta_3}{3}+\frac{109}8-\frac{7\pi^2}8\right)\\ 
        + \epsilon^2\!\left(-\frac{25\zeta_3}2+\frac{639}{16}-\frac{133\pi^2}{48}-\frac{71\pi^4}{1440}\right) + \mathcal O\!\left(\epsilon^3\right).
    \end{gathered}
\end{equation}

\begin{align*}

\subsection{$A_3^1$}

Running \texttt{BuildAntenna[A,3,1,Component->Leading]}, we get the built result,
\begin{align*}
    A_3^1 = {} &\frac1{2(\epsilon-1)q^2}\Bigg((-s_{13})^{-\epsilon}C_\epsilon A_3^{1,(s_{13})}+(-s_{23})^{-\epsilon}C_\epsilon A_3^{1,(s_{23})}+\!\left(-q^2\right)^{-\epsilon}C_\epsilon A_3^{1,\left(q^2\right)}\\

\end{align*}
where,
\begin{gather*}
    A_3^{1,(s_{13})} = \mathcal F_1(s_{12};s_{13},s_{23}),\quad A_3^{1,(s_{23})} = \mathcal F_1(s_{12};s_{23},s_{13})
\end{gather*}
\begin{align*}
    \mathcal F_1(s_{12};x,y) = {} &\frac{1}{4\!\left(q^2\right)^2(s_{12}+y)^2}\Bigg[\frac1\epsilon\left(10s_{12}^2+4s_{12}(x+4y)+3y(x+2y)\right)\\

\end{align*}
\begin{gather*}
    A_3^{1,\left(q^2\right)} = \frac{\mathcal P\left(20s_{12}^2\!\left(q^2\right)^2+28s_{12}q^2s_{13}s_{23}+11s_{13}^2s_{23}^2\right)}{12\epsilon\!\left(q^2\right)^2s_{13}s_{23}\left(s_{12}q^2+{13}s_{23}\right)^2}+\mathcal O(\epsilon),\quad \mathcal P = \left(q^2\right)^2+s_{12}^2-2s_{13}s_{23}
\end{gather*}
\begin{gather*}
    A_3^{1,(\text{Trans.})} = -\frac1{2q^2}A_3^0.
\end{gather*}

\end{equation}

\end{align*}
\begin{equation*}
    \mathcal R(\epsilon,s_{13},s_{23}) = \frac{C_\epsilon}{\epsilon^2(2\epsilon-1)s_{13}s_{23}}

\begin{align*}

    &+5x^2y\bigg)+\epsilon^4(s_{12}+y)(x+y)\bigg((s_{12}+y)(4x-y)+2xy\bigg)    
\end{align*}
\begin{align*}
    \tilde A_3^{1,\left(q^2\right)} = {} &2s_{12}^5\!\left(3u^2-2v\right)+12s_{12}^4u\!\left(u^2-v\right)+s_{12}^3\left(9u^4-10u^3v-8v^2\right)+s_{12}^2u\left(3u^4-16v^2\right)\\
